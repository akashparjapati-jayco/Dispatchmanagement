/** * JAYCO Dispatch Board — Google Sheets backend (standalone script version) * -------------------------------------------------------------------- * This script saves/loads the dashboard's data as one JSON blob into a * specific Google Sheet, identified by its Sheet ID below. * * SETUP (do this in order): * 1. Create a blank Google Sheet at sheets.google.com, name it * "JAYCO Dispatch Data". * 2. Look at its URL: https://docs.google.com/spreadsheets/d/SHEET_ID_HERE/edit * Copy the long ID string between /d/ and /edit. * 3. Paste that ID into the SHEET_ID constant below, replacing * "1ffHXSuzhsEx_Vn-MY3GYjePvTEeC5KulogUlDkp11Zc". * 4. Save this script (Ctrl+S). * 5. Click Deploy > New deployment > Web app. * Execute as: Me * Who has access: Anyone * 6. Deploy, authorize permissions when asked. * 7. Copy the Web App URL it gives you and send it back so the * dashboard can be connected to it. */ const SHEET_ID = "1ffHXSuzhsEx_Vn-MY3GYjePvTEeC5KulogUlDkp11Zc"; const SHEET_TAB_NAME = "DispatchData"; function getSheet_() { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_TAB_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_TAB_NAME); sheet.getRange("A1").setValue("Dashboard JSON (do not edit manually)"); sheet.getRange("B1").setValue("Last Updated"); } return sheet; } function doGet(e) { const sheet = getSheet_(); const json = sheet.getRange("A2").getValue(); const output = json && json.toString().length > 0 ? json.toString() : ""; return ContentService .createTextOutput(output) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const sheet = getSheet_(); const body = e.postData.contents; sheet.getRange("A2").setValue(body); sheet.getRange("B2").setValue(new Date()); return ContentService .createTextOutput(JSON.stringify({ status: "ok" })) .setMimeType(ContentService.MimeType.JSON); }
